# Importing Libraries
import pandas as pd

# Realistic IPL Sample Data
data = {
    'season': [2023, 2023, 2022, 2022, 2021, 2021, 2020, 2020, 2019],
    'team1': ['MI', 'RCB', 'CSK', 'DC', 'KKR', 'SRH', 'RR', 'PBKS', 'GT'],
    'team2': ['CSK', 'MI', 'RCB', 'MI', 'DC', 'RCB', 'MI', 'KKR', 'LSG'],
    'winner': ['CSK', 'RCB', 'CSK', 'MI', 'DC', 'RCB', 'MI', 'KKR', 'GT'],
    'city': ['Mumbai', 'Bangalore', 'Chennai', 'Delhi', 'Kolkata', 'Hyderabad', 'Jaipur', 'Ahmedabad', 'Lucknow'],
    'player_of_match': ['Ruturaj Gaikwad', 'Faf du Plessis', 'Moeen Ali', 'Suryakumar Yadav', 'Prithvi Shaw',
                        'Glenn Maxwell', 'Jasprit Bumrah', 'Andre Russell', 'Shubman Gill'],
    'runs_scored': [218, 190, 205, 198, 210, 187, 199, 202, 215],
}

# Create DataFrame
ipl_df = pd.DataFrame(data)

# 1. Basic Exploration
print("Data Info:")
ipl_df.info()
print("\nFirst 5 Rows:")
print(ipl_df.head())

# 2. Team Performance
team_wins = ipl_df['winner'].value_counts()
print("\nTeam Wins:\n", team_wins)

# 3. Player of the Match Counts
player_of_match_counts = ipl_df['player_of_match'].value_counts()
print("\nPlayer of the Match Counts:\n", player_of_match_counts)

# 4. Average Runs Analysis
average_runs_per_match = ipl_df['runs_scored'].mean()
print(f"\nAverage Runs Scored per Match: {average_runs_per_match:.1f}")
